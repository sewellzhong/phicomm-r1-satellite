cmake_minimum_required(VERSION 3.18)
project(r1_vad LANGUAGES C CXX)
if(NOT DEFINED FVAD_DIR)
  message(FATAL_ERROR "FVAD_DIR is required")
endif()
add_subdirectory("${FVAD_DIR}" fvad-build)
add_library(r1_vad SHARED vad_jni.cpp)
target_link_libraries(r1_vad PRIVATE fvad)
target_compile_features(r1_vad PRIVATE cxx_std_17)
target_compile_options(r1_vad PRIVATE -Wall -Wextra)
target_link_options(r1_vad PRIVATE "-Wl,--hash-style=both")
