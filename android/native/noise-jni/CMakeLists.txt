cmake_minimum_required(VERSION 3.18)
project(r1_noise LANGUAGES C CXX)
add_compile_definitions(NOISE_USE_REFERENCE_BACKEND=1 NOISE_USE_LIBSODIUM=0 NOISE_USE_CUSTOM_RAND=0)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${NOISE_DIR}" noise-build)
add_library(r1_noise SHARED noise_jni.cpp)
target_link_libraries(r1_noise PRIVATE noise_c)
target_compile_features(r1_noise PRIVATE cxx_std_17)
target_link_options(r1_noise PRIVATE "-Wl,--hash-style=both")
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(r1_noise PRIVATE ${JNI_INCLUDE_DIRS})
endif()
